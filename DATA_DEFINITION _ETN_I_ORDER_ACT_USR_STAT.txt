define view entity /ETN/I_ORDER_ACT_USR_STAT
                           as select from jsto
                                      inner join jest
                                              on jsto.objnr = jest.objnr
                                             and jsto.obtyp = 'ORI'         // ObjectType - Auftrag
                                             and jsto.stsma like 'ZDPMZD%'  // Statusschema zu unseren Auftragsarten - siehe T003o
                                             and jest.stat  like 'E%'        // nur Anwenderstatus
                                             and jest.inact = ''             // nur aktive Status
                                             and jest.objnr like 'OR%'       // nur Auftrags-Objekte

{
 key jest.objnr     as ObjectKey,
 key jest.stat      as Status,
     jsto.stsma     as StatusProfile
}
